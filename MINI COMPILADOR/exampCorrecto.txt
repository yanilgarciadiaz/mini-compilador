// Programa correcto en las tres fases (sin errores ni advertencias).
int contador = 0;
float total = 0.0;

for (int i = 0; i < 5; i = i + 1) {
    total = total + i * 1.5;
    contador = contador + 1;
}

while (contador > 0) {
    contador = contador - 1;
}

if (total >= 10.0) {
    print("Total alto", total);
} else {
    print("Total bajo", total);
}
